    <script>

        const API_URL = 'https://script.google.com/macros/s/AKfycbzMGV1nL9tpP4fm7P3oeHbVEmsoWZfpXlLriGJ8hsY3uGFvwBkqSic90ZBJaqAA9NRH/exec';

        

        async function loadLeaderboard() {

            try {

                const response = await fetch(API_URL + '?t=' + Date.now());

                const data = await response.json();

                

                if (data.success && data.data) {

                    const players = data.data;

                    const totalPlayers = players.length;

                    const totalMW = players.reduce((sum, p) => sum + p.taps, 0);

                    

                    document.getElementById('totalPlayers').textContent = totalPlayers;

                    document.getElementById('totalMW').textContent = totalMW;

                    

                    // Подиум

                    let podiumHtml = '';

                    if (players.length > 1) podiumHtml += `<div class="podium-card podium-2"><div class="podium-rank">2</div><div class="podium-name">${players[1]?.userId?.substring(0,8)}</div><div class="podium-score">${players[1]?.taps} <small>МВт</small></div></div>`;

                    if (players.length > 0) podiumHtml += `<div class="podium-card podium-1"><div class="crown">👑</div><div class="podium-rank">1</div><div class="podium-name">${players[0]?.userId?.substring(0,8)}</div><div class="podium-score">${players[0]?.taps} <small>МВт</small></div></div>`;

                    if (players.length > 2) podiumHtml += `<div class="podium-card podium-3"><div class="podium-rank">3</div><div class="podium-name">${players[2]?.userId?.substring(0,8)}</div><div class="podium-score">${players[2]?.taps} <small>МВт</small></div></div>`;

                    document.getElementById('podium').innerHTML = podiumHtml;

                    

                    // Таблица

                    const rowsHtml = players.slice(3, 20).map((player, index) => `

                        <div class="table-row">

                            <div class="rank">${index + 4}</div>

                            <div class="user-info"><div class="avatar">⚡</div><span class="username">${player.userId?.substring(0,12)}...</span></div>

                            <div class="score">${player.taps}</div>

                            <div class="trend">📈 ${Math.floor(Math.random()*10)+1}%</div>

                        </div>

                    `).join('');

                    

                    document.getElementById('leaderboardRows').innerHTML = rowsHtml || '<div style="text-align:center;padding:30px;color:#8096c0">Нет данных</div>';

                    document.getElementById('updateTime').innerHTML = `<i class="far fa-clock"></i> Обновлено: ${new Date().toLocaleTimeString()}`;

                }

            } catch(error) {

                document.getElementById('leaderboardRows').innerHTML = `<div style="color:#ff6b6b;padding:30px;text-align:center">Ошибка загрузки</div>`;

            }

        }

        

        loadLeaderboard();

        setInterval(loadLeaderboard, 5000);

    </script>
