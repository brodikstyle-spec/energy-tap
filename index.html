    <script>

        // === НАСТРОЙКИ === 

        const API_URL = 'https://script.google.com/macros/s/https://script.google.com/macros/s/AKfycbzMGV1nL9tpP4fm7P3oeHbVEmsoWZfpXlLriGJ8hsY3uGFvwBkqSic90ZBJaqAA9NRH/exec';  // ⚡ ВСТАВЬТЕ СЮДА URL ИЗ ШАГА 2!

        const SAVE_INTERVAL = 5; // сохранять каждые 5 тапов

        // =================

        

        // Генерируем или получаем ID пользователя

        let userId = localStorage.getItem('energyTap_userId');

        if (!userId) {

            userId = 'user_' + Date.now() + '_' + Math.random().toString(36).substring(2, 8);

            localStorage.setItem('energyTap_userId', userId);

        }

        

        // Пытаемся получить имя пользователя (можно попросить ввести)

        let userName = localStorage.getItem('energyTap_userName') || 'Участник';

        

        // Загружаем счетчик из localStorage

        let counter = parseInt(localStorage.getItem('energyTap_counter_' + userId)) || 0;

        let lastSavedCounter = counter;

        

        // Обновляем отображение

        document.getElementById('counter').innerText = counter;

        document.getElementById('userId').innerText = userId.substring(0, 12) + '...';

        

        // Функция отправки данных на сервер

        async function sendTapsToServer(tapsValue) {

            const statusEl = document.getElementById('statusMessage');

            statusEl.innerText = '⏳ Отправка данных...';

            statusEl.className = 'status-message';

            

            try {

                // Получаем информацию о браузере

                const userAgent = navigator.userAgent.substring(0, 100);

                

                // Создаем уникальный ID запроса для предотвращения дубликатов

                const requestId = Date.now() + '_' + Math.random();

                

                const response = await fetch(API_URL, {

                    method: 'POST',

                    mode: 'no-cors', // важно для Google Apps Script

                    headers: {

                        'Content-Type': 'application/json',

                    },

                    body: JSON.stringify({

                        userId: userId,

                        userName: userName,

                        taps: tapsValue,

                        userAgent: userAgent,

                        requestId: requestId,

                        timestamp: new Date().toISOString()

                    })

                });

                

                // Из-за no-cors мы не можем прочитать ответ, но знаем что запрос ушел

                lastSavedCounter = tapsValue;

                localStorage.setItem('energyTap_counter_' + userId, tapsValue);

                

                statusEl.innerText = '✅ Данные сохранены в облаке! Всего: ' + tapsValue + ' МВт';

                statusEl.className = 'status-message success';

                

                // Обновляем время последнего сохранения

                localStorage.setItem('energyTap_lastSave_' + userId, Date.now());

                

            } catch(error) {

                console.error('Ошибка отправки:', error);

                statusEl.innerText = '❌ Ошибка соединения, но данные сохранены локально';

                statusEl.className = 'status-message error';

            }

        }

        

        // Обработка тапа

        function handleTap() {

            // Увеличиваем счетчик

            counter++;

            

            // Мгновенно обновляем интерфейс

            document.getElementById('counter').innerText = counter;

            

            // Сохраняем в localStorage сразу

            localStorage.setItem('energyTap_counter_' + userId, counter);

            

            // Виброотклик (если поддерживается)

            if (navigator.vibrate) {

                navigator.vibrate(20);

            }

            

            // Анимация кнопки

            const btn = document.getElementById('tapBtn');

            btn.style.transform = 'scale(0.92)';

            setTimeout(() => {

                btn.style.transform = 'scale(1)';

            }, 80);

            

            // Отправляем на сервер с определенной периодичностью

            if (counter % SAVE_INTERVAL === 0) {

                sendTapsToServer(counter);

            } else {

                // Показываем статус

                document.getElementById('statusMessage').innerText = `⚡ +1 МВт! Всего: ${counter}`;

                document.getElementById('statusMessage').className = 'status-message';

            }

        }

        

        // При загрузке отправляем текущее значение на сервер

        window.addEventListener('load', () => {

            if (counter > 0) {

                setTimeout(() => sendTapsToServer(counter), 1000);

            }

            

            // Периодическая синхронизация (каждые 30 секунд)

            setInterval(() => {

                if (counter > lastSavedCounter) {

                    sendTapsToServer(counter);

                }

            }, 30000);

        });

        

        // Сохраняем при закрытии вкладки

        window.addEventListener('beforeunload', () => {

            if (counter > lastSavedCounter) {

                // Используем sendBeacon для надежности

                const data = JSON.stringify({

                    userId: userId,

                    taps: counter,

                    userName: userName,

                    timestamp: new Date().toISOString()

                });

                navigator.sendBeacon(API_URL, data);

            }

        });

    </script>
